Sekarang latihan terakhir Hari 3.

```
func validateQuantity(
    quantity: Int
) -> String? {

    guard quantity > 0 else {
        return "Jumlah minimal 1"
    }

    guard quantity <= 10 else {
        return "Jumlah maksimal 10"
    }

    return nil
}

func calculateTotal(
    price: Int,
    quantity: Int
) -> Int {
    return price * quantity
}

func processPurchase(
    price: Int,
    quantity: Int
) -> String {

    if let error = validateQuantity(
        quantity: quantity
    ) {
        return error
    }

    let total = calculateTotal(
        price: price,
        quantity: quantity
    )

    return "Total: \(total)"
}
```
